# Builds hlslcc_cli: a batch DXBC -> Metal translator on top of Unity's HLSLcc.
cmake_minimum_required(VERSION 3.14)
project(hlslcc_cli CXX C)
include(FetchContent)
FetchContent_Declare(hlslcc GIT_REPOSITORY https://github.com/Unity-Technologies/HLSLcc.git GIT_TAG 3ea1fcd)
FetchContent_GetProperties(hlslcc)
if(NOT hlslcc_POPULATED)
  FetchContent_Populate(hlslcc)
endif()
file(GLOB HLSLCC_SRC ${hlslcc_SOURCE_DIR}/src/*.cpp ${hlslcc_SOURCE_DIR}/src/cbstring/*.c)
add_library(hlslcc STATIC ${HLSLCC_SRC})
add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
if(MSVC)
  add_compile_options(/EHa /Zi)
endif()
target_include_directories(hlslcc PUBLIC ${hlslcc_SOURCE_DIR} ${hlslcc_SOURCE_DIR}/include ${hlslcc_SOURCE_DIR}/src ${hlslcc_SOURCE_DIR}/src/internal_includes ${hlslcc_SOURCE_DIR}/src/cbstring)
add_executable(hlslcc_tool main.cpp)
target_link_libraries(hlslcc_tool hlslcc dbghelp)
if(MSVC)
  target_compile_options(hlslcc_tool PRIVATE /Zi)
  target_link_options(hlslcc_tool PRIVATE /DEBUG)
endif()
